def find_primes(n: int) -> list[int]:
    """
    找出1到n之间所有的素数。

    参数:
        n (int): 查找素数的上限值（包含n）。

    返回:
        list[int]: 包含1到n之间所有素数的列表。

    异常:
        ValueError: 当 n 小于 1 时抛出。
    """
    if n < 1:
        raise ValueError("n 必须是大于或等于 1 的正整数。")

    if n < 2:
        return []

    is_prime = [True] * (n + 1)
    is_prime[0] = is_prime[1] = False

    for p in range(2, int(n**0.5) + 1):
        if is_prime[p]:
            for i in range(p * p, n + 1, p):
                is_prime[i] = False

    return [p for p in range(2, n + 1) if is_prime[p]]

**时间复杂度分析**
当前代码采用埃拉托斯特尼筛法（Sieve of Eratosthenes），时间复杂度为 $O(n \log \log n)$。在时间复杂度层面，该算法已达到理论最优（阿特金筛法虽理论复杂度为 $O(n)$，但常数因子过大，实际运行效率通常不及埃氏筛）。

**算法效率的主要瓶颈**
1. **空间复杂度瓶颈**：空间复杂度为 $O(n)$。当 $n$ 较大（如 $>10^8$）时，`[True] * (n + 1)` 会消耗大量内存，极易导致内存溢出（OOM）。
2. **常数时间瓶颈（Python解释器开销）**：内层标记合数的操作使用了Python原生的 `for` 循环，在解释器层面执行存在巨大的常数时间开销；同时，`list` 存储布尔值的内存布局不紧凑，导致CPU缓存命中率低。

**改进算法与优化建议**
1. **算法级空间优化（分段筛法 Segmented Sieve）**：
   引入**分段筛法**。先求出 $\sqrt{n}$ 以内的基础素数，然后将 $[1, n]$ 划分为多个长度为 $\sqrt{n}$ 的区间进行分段筛选。此改进可将空间复杂度从 $O(n)$ 骤降至 $O(\sqrt{n})$，彻底解决大 $n$ 场景下的内存瓶颈。
2. **算法级常数优化（奇数筛法 Odd-only Sieve）**：
   除2以外，所有偶数均非素数。可单独处理2，随后仅对奇数建立筛法数组。此改进可将时间和空间复杂度的常数因子直接减半。
3. **工程级向量化优化（针对Python特性）**：
   - 使用 `bytearray` 或位图（Bitset）替代 `list`，大幅压缩内存并提升缓存命中率。
   - 利用Python的**切片赋值**（如 `is_prime[p*p::p] = bytearray(len(range(p*p, n+1, p)))`）替代内层 `for` 循环，将循环开销下沉至C底层，可消除Python解释器的循环瓶颈，获得显著的性能提升。

def find_primes(n: int) -> list[int]:
    """
    找出1到n之间所有的素数。

    参数:
        n (int): 查找素数的上限值（包含n）。

    返回:
        list[int]: 包含1到n之间所有素数的列表。

    异常:
        ValueError: 当 n 小于 1 时抛出。
    """
    if n < 1:
        raise ValueError("n 必须是大于或等于 1 的正整数。")

    if n < 2:
        return []

    limit = int(n**0.5)

    # 基础筛法 (Base Sieve) - 仅处理奇数，空间复杂度 O(sqrt(n))
    base_size = (limit - 1) // 2
    is_prime_base = bytearray([1]) * base_size

    for i in range(base_size):
        if is_prime_base[i]:
            p = 2 * i + 3
            start_idx = (p * p - 3) // 2
            # 使用 bytes(length) 生成全0字节串，利用C底层循环进行切片赋值，消除Python解释器开销
            is_prime_base[start_idx::p] = bytes(len(range(start_idx, base_size, p)))

    base_primes = [2]
    for i in range(base_size):
        if is_prime_base[i]:
            base_primes.append(2 * i + 3)

    primes = list(base_primes)

    # 分段筛法 (Segmented Sieve) - 将区间划分为长度为 sqrt(n) 的段
    segment_size = max(int(n**0.5), 1)
    low = limit + 1
    if low % 2 == 0:
        low += 1

    while low <= n:
        high = min(low + segment_size - 1, n)
        if high % 2 == 0:
            high -= 1

        if low > high:
            break

        seg_size = (high - low) // 2 + 1
        is_prime_seg = bytearray([1]) * seg_size

        for p in base_primes:
            # 当 p^2 > high 时，后续素数的平方必然大于 high，可直接提前终止内层循环
            if p * p > high:
                break
            if p == 2:
                continue

            # 找到段内第一个能被 p 整除的奇数
            start = ((low + p - 1) // p) * p
            if start % 2 == 0:
                start += p

            if start < p * p:
                start = p * p

            if start > high:
                continue

            start_idx = (start - low) // 2
            is_prime_seg[start_idx::p] = bytes(len(range(start_idx, seg_size, p)))

        # 收集当前段内的素数
        for i in range(seg_size):
            if is_prime_seg[i]:
                primes.append(low + 2 * i)

        low = high + 2

    return primes

def find_primes(n: int) -> list[int]:
    """
    找出1到n之间所有的素数。

    参数:
        n (int): 查找素数的上限值（包含n）。

    返回:
        list[int]: 包含1到n之间所有素数的列表。

    异常:
        ValueError: 当 n 小于 1 时抛出。
    """
    if n < 1:
        raise ValueError("n 必须是大于或等于 1 的正整数。")
    if n < 2:
        return []

    # Mod 30 轮筛法 (Wheel Factorization) 常量
    WHEEL = [1, 7, 11, 13, 17, 19, 23, 29]
    MOD30_TO_IDX = [-1] * 30
    for i, w in enumerate(WHEEL):
        MOD30_TO_IDX[w] = i

    # 预计算 WHEEL 中小于 i 的元素个数，用于 O(1) 计算段大小
    WHEEL_COUNT = [0] * 31
    for i in range(1, 31):
        WHEEL_COUNT[i] = sum(1 for w in WHEEL if w < i)

    def get_size(max_val: int) -> int:
        """计算 [1, max_val] 中与 30 互素的整数个数"""
        if max_val < 30:
            return WHEEL_COUNT[max_val + 1]
        q, r = divmod(max_val, 30)
        return q * 8 + WHEEL_COUNT[r + 1]

    limit = int(n**0.5)
    base_size = get_size(limit)
    is_prime_base = bytearray([1]) * base_size

    if base_size > 0:
        is_prime_base[0] = 0  # 1 不是素数

    # 基础筛法 (Base Sieve)
    for i in range(base_size):
        if is_prime_base[i]:
            q, r = divmod(i, 8)
            p = 30 * q + WHEEL[r]

            if p * p > limit:
                break

            step = 8 * p
            for w in WHEEL:
                k = (p - w + 29) // 30
                start_val = p * (w + 30 * k)
                if start_val > limit:
                    continue
                start_idx = (start_val // 30) * 8 + MOD30_TO_IDX[(p * w) % 30]
                length = (base_size - 1 - start_idx) // step + 1
                is_prime_base[start_idx::step] = bytes(length)

    base_primes = [2, 3, 5]
    for i in range(base_size):
        if is_prime_base[i]:
            q, r = divmod(i, 8)
            base_primes.append(30 * q + WHEEL[r])

    primes = [p for p in base_primes if p <= n]

    # 分段筛法 (Segmented Sieve)
    # 固定段大小为 32KB (32768 bytes)，以完美适配 L1 数据缓存
    SEGMENT_SIZE = 32768

    low = limit + 1
    # 将 low 对齐到 30 的倍数，简化段内索引计算
    low = ((low + 29) // 30) * 30

    while low <= n:
        num_blocks = SEGMENT_SIZE // 8
        high = low + 30 * num_blocks - 1

        if high > n:
            num_blocks = (n - low + 1) // 30
            if num_blocks == 0:
                high = n
            else:
                high = low + 30 * num_blocks - 1

        full_blocks = (high - low + 1) // 30
        rem = (high - low + 1) % 30
        seg_size = full_blocks * 8 + WHEEL_COUNT[rem]

        is_prime_seg = bytearray([1]) * seg_size

        for p in base_primes:
            if p * p > high:
                break
            if p in (2, 3, 5):
                continue

            step = 8 * p
            for w in WHEEL:
                target_mod = p * w
                mod_val = 30 * p

                rem_low = low % mod_val
                if rem_low <= target_mod:
                    start_val = low - rem_low + target_mod
                else:
                    start_val = low - rem_low + mod_val + target_mod

                if start_val > high:
                    continue

                w_prime = target_mod % 30
                idx_w_prime = MOD30_TO_IDX[w_prime]
                start_idx = ((start_val - low - w_prime) // 30) * 8 + idx_w_prime

                length = (seg_size - 1 - start_idx) // step + 1
                is_prime_seg[start_idx::step] = bytes(length)

        # 收集当前段内的素数
        for i in range(seg_size):
            if is_prime_seg[i]:
                q, r = divmod(i, 8)
                primes.append(low + 30 * q + WHEEL[r])

        # 移动到下一段
        low = high + 1
        low = ((low + 29) // 30) * 30

    return primes
