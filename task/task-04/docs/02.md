*示例 1：单机数据清洗 (Pandas)*
```python
import pandas as pd
import numpy as np

def clean_and_transform_data(file_path: str) -> pd.DataFrame:
    """读取CSV并进行基础数据清洗"""
    try:
        # 1. 提取 (Extract)
        df = pd.read_csv(file_path, parse_dates=['timestamp'])

        # 2. 转换 (Transform)
        # 去除重复行
        df = df.drop_duplicates(subset=['user_id', 'timestamp'])
        # 处理缺失值：数值型填0，分类型填'Unknown'
        df['amount'] = df['amount'].fillna(0)
        df['category'] = df['category'].fillna('Unknown')
        # 业务逻辑转换
        df['is_high_value'] = np.where(df['amount'] > 1000, 1, 0)

        return df
    except Exception as e:
        print(f"数据处理失败: {e}")
        raise
```

*示例 2：分布式ETL处理 (PySpark)*
```python
from pyspark.sql import SparkSession
from pyspark.sql.functions import col, when

def spark_etl_pipeline(input_path: str, output_path: str):
    """使用PySpark进行分布式数据ETL"""
    # 1. 初始化SparkSession
    spark = SparkSession.builder \
        .appName("DataEngineeringETL") \
        .config("spark.sql.shuffle.partitions", "200") \
        .getOrCreate()

    try:
        # 2. 读取数据 (支持Parquet/JSON/CSV等)
        df = spark.read.parquet(input_path)

        # 3. 数据转换与过滤
        df_transformed = df.filter(col("status") == "ACTIVE") \
            .withColumn("discounted_price",
                        when(col("price") > 100, col("price") * 0.9)
                        .otherwise(col("price")))

        # 4. 加载数据 (写入分区表)
        df_transformed.write \
            .mode("overwrite") \
            .partitionBy("year", "month") \
            .parquet(output_path)

    finally:
        spark.stop()

# 执行

```

*示例 3：数据库交互与批量写入 (SQLAlchemy)*
```python
import sqlalchemy
from sqlalchemy import create_engine, text
import pandas as pd

def load_data_to_db(df: pd.DataFrame, db_url: str, table_name: str):
    """将Pandas DataFrame批量写入关系型数据库"""
    engine = create_engine(db_url, pool_pre_ping=True)

    try:
        with engine.begin() as conn: # 使用事务保证原子性
            # 清空目标表（可选）
            conn.execute(text(f"TRUNCATE TABLE {table_name}"))

            # 批量写入数据
            df.to_sql(
                name=table_name,
                con=conn,
                if_exists='append',
                index=False,
                chunksize=10000 # 分块写入防止内存溢出
            )
        print(f"成功写入 {len(df)} 条数据到 {table_name}")
    except Exception as e:
        print(f"数据库写入失败: {e}")
        raise
    finally:
        engine.dispose()
```
