```python
"""
API客户端模块
用于与外部API交互
"""

import logging
import time
import requests
from typing import Dict, Any, Optional, Callable
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry

logger = logging.getLogger(__name__)


class APIError(Exception):
    """API错误基类"""
    def __init__(self, message: str, status_code: Optional[int] = None, response: Optional[Dict] = None):
        super().__init__(message)
        self.status_code = status_code
        self.response = response


class APIClient:
    """API客户端基类"""

    def __init__(
        self,
        base_url: str,
        api_key: Optional[str] = None,
        timeout: int = 30,
        max_retries: int = 3,
        retry_delay: float = 1.0
    ):
        """
        初始化API客户端

        Args:
            base_url: API基础URL
            api_key: API密钥
            timeout: 请求超时时间（秒）
            max_retries: 最大重试次数
            retry_delay: 重试间隔（秒）
        """
        self.base_url = base_url.rstrip('/')  # 移除末尾斜杠
        self.api_key = api_key
        self.timeout = timeout
        self.max_retries = max_retries
        self.retry_delay = retry_delay

        # 配置session和重试策略
        self.session = requests.Session()
        self._setup_retry()

        if api_key:
            self.session.headers.update({
                'Authorization': f'Bearer {api_key}'
            })

    def _setup_retry(self):
        """配置重试策略"""
        retry_strategy = Retry(
            total=self.max_retries,
            backoff_factor=self.retry_delay,
            status_forcelist=[429, 500, 502, 503, 504],  # 可重试的状态码
            allowed_methods=["GET", "POST", "PUT", "DELETE"],
            raise_on_status=False
        )
        adapter = HTTPAdapter(max_retries=retry_strategy)
        self.session.mount("http://", adapter)
        self.session.mount("https://", adapter)

    def _make_request(
        self,
        method: str,
        endpoint: str,
        **kwargs
    ) -> Dict[str, Any]:
        """
        统一的请求处理方法

        Args:
            method: HTTP方法
            endpoint: API端点
            **kwargs: 传递给requests的参数

        Returns:
            响应数据

        Raises:
            APIError: API请求失败
        """
        url = f"{self.base_url}/{endpoint.lstrip('/')}"
        kwargs.setdefault('timeout', self.timeout)

        logger.debug(f"Sending {method} request to {url}")

        try:
            response = self.session.request(method, url, **kwargs)
            response.raise_for_status()

            # 验证响应格式
            if response.content:
                try:
                    return response.json()
                except requests.exceptions.JSONDecodeError as e:
                    raise APIError(
                        f"Invalid JSON response: {e}",
                        status_code=response.status_code
                    )
            return {}

        except requests.exceptions.ConnectionError as e:
            logger.error(f"Connection error: {e}")
            raise APIError(f"Connection failed: {e}")

        except requests.exceptions.Timeout as e:
            logger.error(f"Request timeout after {self.timeout}s: {e}")
            raise APIError(f"Request timeout after {self.timeout}s")

        except requests.exceptions.HTTPError as e:
            status_code = e.response.status_code if e.response else None
            logger.error(f"HTTP error {status_code}: {e}")
            raise APIError(
                f"HTTP error: {e}",
                status_code=status_code,
                response=e.response.json() if e.response else None
            )

        except requests.exceptions.RequestException as e:
            logger.error(f"Request failed: {e}")
            raise APIError(f"Request failed: {e}")

    def get(self, endpoint: str, params: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
        """发送GET请求"""
        return self._make_request("GET", endpoint, params=params)

    def post(self, endpoint: str, data: Dict[str, Any]) -> Dict[str, Any]:
        """发送POST请求"""
        return self._make_request("POST", endpoint, json=data)

    def put(self, endpoint: str, data: Dict[str, Any]) -> Dict[str, Any]:
        """发送PUT请求"""
        return self._make_request("PUT", endpoint, json=data)

    def delete(self, endpoint: str, confirm: bool = False) -> None:
        """
        发送DELETE请求

        Args:
            endpoint: API端点
            confirm: 是否确认删除（安全保护）
        """
        if not confirm:
            raise ValueError("DELETE operation requires confirm=True")

        self._make_request("DELETE", endpoint)
        logger.info(f"Successfully deleted resource: {endpoint}")
